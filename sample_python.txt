def verify_active_profile(page, target_name=TARGET_PROFILE_NAME):
    """
    Navigates to LinkedIn and verifies the active account-specific identity.
    Stops execution if LinkedIn is logged in under a different account or unauthenticated.
    """
    print(f"[*] Verifying that active browser session belongs to '{target_name}'...")
    page.goto("https://www.linkedin.com/feed/", wait_until="domcontentloaded", timeout=30000)
    time.sleep(3)
    # Check for login redirection
    if any(auth in page.url.lower() for auth in ["/login", "/authwall", "/checkpoint"]):
        print("\n[!] ERROR: LinkedIn session is not logged in.")
        print("[!] Please log into LinkedIn in Edge and re-run.\n")
        sys.exit(1)
    # Check identity card link in sidebar
    profile_links = page.query_selector_all('a[href*="/in/"]')
    for pl in profile_links:
        txt = pl.inner_text().strip()
        href = pl.get_attribute("href") or ""
        # Verify name match
        if target_name.lower() in txt.lower():
            first_line = txt.splitlines()[0] if txt else target_name
            print(f"[✓] CONFIRMED: Active LinkedIn identity verified as '{first_line}' ({href})")
            return True
    # Fallback check on /in/me/
    try:
        page.goto("https://www.linkedin.com/in/me/", wait_until="domcontentloaded", timeout=15000)
        time.sleep(2)
        if target_name.lower() in page.title().lower():
            print(f"[✓] CONFIRMED: Active profile verified via /in/me/ title: '{page.title()}'")
            return True
    except Exception:
        pass
    print(f"\n[!] ERROR: Current logged-in LinkedIn user does not match '{target_name}'.")
    print(f"[!] Intended profile '{target_name}' could not be verified on LinkedIn.")
    print("[!] Execution stopped to prevent scraping under a different account.\n")
    sys.exit(1)
